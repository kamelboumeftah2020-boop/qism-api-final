function genUsername(role){const num=Math.floor(10000000+Math.random()*90000000); return (role==='teacher'?'ostad_':role==='parent'?'wali_':'tilmid_')+num;}
function genSlug(display){return display.toLowerCase().replace(/[^a-z0-9\u0600-\u06FF]+/g,'-').replace(/^-|-$/g,'').slice(0,30)+'-'+Math.floor(100000+Math.random()*900000);}
function genToken(u){const exp=Date.now()+3600000; return Buffer.from(JSON.stringify({id:u.id,email:u.email,role:u.role,name:u.display_name||u.name,username:u.username,exp:exp})).toString('base64');}
function getUser(req){try{let t=(req.headers.authorization||'').replace('Bearer ','')||req.query.token; if(!t) return null; const p=JSON.parse(Buffer.from(t,'base64').toString()); if(p.exp&&p.exp<Date.now()) return null; return p;}catch{return null;}}
async function calcComm(price,completed){let rate=completed>80?15:30; let fix=200; return {rate,commission:Math.floor(price*rate/100),fixed:fix,teacherEarning:price-Math.floor(price*rate/100)-fix,platformFee:Math.floor(price*rate/100)+fix};}
async function findOrCreateUser(email,display,role,prov,pid,avatar){
 const clean=(email||'').toLowerCase(); if(!pool) return {id:1,email:clean,display_name:display,username:genUsername(role),slug:genSlug(display),role:role};

}
